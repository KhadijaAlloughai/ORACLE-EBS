-First create procedure in oracle EBS should be in this way :
   1.MANDATORY PARAMETERS
        a. ERRBUF OUT VARCHAR2
        b. RETCODE OUT NUMBER
   2. FND_FILE.PUT_LINE(FND_OUTPUT, ‘ ORACLE EBS PROCEDURE !!’);


SQL> CREATE OR REPLACE PROCEDURE KHADIJA_PR_EBS(ERRBUF OUT VARCHAR2,
RETCODE OUT NUMBER) IS
BEGIN
FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ‘ WELCOME TO ORACLE EBS PROC !!!’);
END;
